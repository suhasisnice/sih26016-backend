# Pinned to the versions this stack was actually built and verified
# against. Unpinned, a deploy months from now resolves whatever is newest
# and a breaking release becomes a deployment-day mystery — the one moment
# there is no time to debug a dependency.
#
# bcrypt is held at 4.0.1 specifically: passlib 1.7.4 reads bcrypt.__about__
# to detect the version, and bcrypt 4.1+ removed it, which makes passlib
# fail on every hash with a confusing AttributeError.

# Schema migrations. Every schema change from Phase 1 onward is a revision
# in alembic/versions; nothing calls create_all() any more. The container
# runs `alembic upgrade head` before uvicorn starts, so a deploy migrates
# itself rather than depending on somebody remembering to.
alembic==1.19.1

# Face login. face_recognition wraps dlib's ResNet face-recognition model —
# see the Dockerfile for why the image needs cmake/g++/make to build it.
# Pulls in dlib>=19.7 and face-recognition-models as transitive deps,
# compiled from source; there is no prebuilt wheel for this base image.
#
# setuptools: face-recognition-models (unmaintained since ~2018) still
# imports pkg_resources. setuptools 81+ dropped it outright, so "just
# install setuptools" isn't enough on its own — confirmed against the
# actual failure ("No module named 'pkg_resources'") with the *latest*
# setuptools installed, not a hypothetical. 80.10.2 is the newest release
# that still carries it; anything 81.0.0+ reintroduces the exact same
# crash on `import face_recognition`.
setuptools==80.10.2
face-recognition==1.3.0
numpy==2.5.2
Pillow==12.3.0

# Authenticator-app second factor for password login. pyotp does the

# with Pillow, already a dependency above, so this adds no new image

pyotp==2.9.0
qrcode==8.0
